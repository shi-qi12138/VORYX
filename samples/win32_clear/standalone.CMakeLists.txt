cmake_minimum_required(VERSION 3.25)
project(VoryxWin32Clear LANGUAGES CXX)
find_package(Voryx 0.1.0 EXACT CONFIG REQUIRED COMPONENTS runtime)

endif()
add_executable(win32_clear main.cpp)
target_link_libraries(win32_clear PRIVATE Voryx::runtime user32)
