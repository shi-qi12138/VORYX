cmake_minimum_required(VERSION 3.25)
project(VoryxXlibClear LANGUAGES CXX)
find_package(Voryx 0.1.0 EXACT CONFIG REQUIRED COMPONENTS runtime)
if(NOT Voryx_WITH_X11)
    message(FATAL_ERROR "The Xlib clear example requires a VORYX SDK with X11 enabled")
endif()
add_executable(xlib_clear main.cpp)
target_link_libraries(xlib_clear PRIVATE Voryx::runtime X11::X11)
