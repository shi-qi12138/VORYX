if(VORYX_ENABLE_WIN32_WSI OR VORYX_ENABLE_X11 OR VORYX_ENABLE_WAYLAND)
    find_package(Vulkan 1.3.280 REQUIRED)
else()
    find_package(Vulkan 1.2 REQUIRED)
endif()
add_library(voryx_backend_vulkan STATIC vulkan_context.cpp vulkan_mapped_buffer.cpp vulkan_buffers.cpp vulkan_graphics_pipeline.cpp vulkan_device_texture.cpp
    vulkan_sampled_texture.cpp vulkan_texture_binding.cpp vulkan_binding_set.cpp)
if(VORYX_ENABLE_WIN32_WSI OR VORYX_ENABLE_X11 OR VORYX_ENABLE_WAYLAND)
    target_sources(voryx_backend_vulkan PRIVATE vulkan_presenter.cpp vulkan_triangle.cpp
        vulkan_presenter_resources.cpp vulkan_depth_target.cpp)
    target_include_directories(voryx_backend_vulkan PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(VORYX_ENABLE_WIN32_WSI)
    target_sources(voryx_backend_vulkan PRIVATE wsi/win32_surface.cpp)
    target_link_libraries(voryx_backend_vulkan PRIVATE user32)
endif()
if(VORYX_ENABLE_X11)
    target_sources(voryx_backend_vulkan PRIVATE wsi/xlib_surface.cpp)
    target_link_libraries(voryx_backend_vulkan PRIVATE X11::X11)
endif()
if(VORYX_ENABLE_WAYLAND)
    target_sources(voryx_backend_vulkan PRIVATE wsi/wayland_surface.cpp)
endif()
target_include_directories(voryx_backend_vulkan PRIVATE "${PROJECT_SOURCE_DIR}/src/rhi")
target_link_libraries(voryx_backend_vulkan PUBLIC Voryx::rhi_api
    PRIVATE voryx_rhi voryx_backend_contract Vulkan::Vulkan)
set_target_properties(voryx_backend_vulkan PROPERTIES EXPORT_NAME detail_vulkan CXX_EXTENSIONS OFF)
voryx_enable_warnings(voryx_backend_vulkan)

target_include_directories(voryx_backend_vulkan PRIVATE "${PROJECT_SOURCE_DIR}/src/core")
